import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

skif combines three kinds of checks that are usually spread over separate tools, for npm, pnpm, yarn and
bun workspaces.

<CardGrid>
    <Card title="Manifest and workspace rules" icon="document">
        Root and member `package.json` rules, workspace pattern checks, sorted and well-placed dependency
        fields.
    </Card>
    <Card title="Version and catalog policies" icon="setting">
        Version groups, semver-range policies, dependency families, workspace protocol and catalog
        enforcement across packages.
    </Card>
    <Card title="Unused and unresolved dependencies" icon="magnifier">
        Static analysis of JavaScript and TypeScript sources on oxc finds unused, unlisted and unresolved
        dependencies, unused files and exports.
    </Card>
    <Card title="Safe autofixes" icon="approve-check">
        Format-preserving edits keep key order, indentation and comments. skif never executes project code.
    </Card>
</CardGrid>
